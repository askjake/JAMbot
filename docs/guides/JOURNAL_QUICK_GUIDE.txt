================================================================================
                              DATABASE QUERIES
================================================================================

# Count journals
SELECT COUNT(*) FROM user_journal;

# Recent journals
SELECT journal_id, owner_id, summary, created_at 
FROM user_journal 
ORDER BY created_at DESC 
LIMIT 10;

# Journals for specific user
SELECT * FROM user_journal 
WHERE owner_id = 'user@example.com' 
ORDER BY created_at DESC;

# Journals for specific chat
SELECT * FROM user_journal 
WHERE chat_id = 'CHAT_UUID_HERE';

# Journal with full analysis
SELECT journal_id, summary, psychoanalysis, interaction_patterns, topics
FROM user_journal
WHERE journal_id = 'JOURNAL_UUID_HERE';

================================================================================
                              API ENDPOINTS
================================================================================

GET /api/journal/my
  - Get user's journals
  - Query params: limit, offset, journal_type
  - Returns: List of journal entries

GET /api/journal/{journal_id}
  - Get specific journal detail
  - Returns: Full journal with all analysis

POST /api/journal/generate/{chat_id}
  - Manually trigger journal generation
  - Returns: Task ID for tracking
